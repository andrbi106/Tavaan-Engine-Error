## استفاده

```php
use Tavaan\Engine\Error\HttpException;
use Tavaan\Engine\Error\ErrorCode;
use Tavaan\Engine\Error\ErrorHandler;
use Tavaan\Engine\Error\ExceptionMapper;
use Tavaan\Engine\Error\LogMasker;
use Tavaan\Engine\Error\RefCodeGenerator;

// تولید خطا
throw HttpException::notFound('صفحه یافت نشد');
throw HttpException::unauthorized();
throw HttpException::forbidden();
throw HttpException::badRequest();
throw HttpException::validation(['name' => 'required']);
throw HttpException::serverError('db failed');
throw HttpException::rateLimited(retryAfter: 60);
throw HttpException::maintenance();

// با refCode
$e = new HttpException(404, 'msg', ErrorCode::NOT_FOUND, [], null, 'ABC12345');

// مدیریت خطا
$handler = new ErrorHandler(debug: true, logger: $logger);
$handler->addSensitiveKeys('custom_secret');
$handler->register();  // set_error_handler, set_exception_handler, shutdown
// ...
$handler->unregister();

// Mapping
$mapper = new ExceptionMapper();
$httpException = $mapper->map(\new \RuntimeException());  // → HttpException(500)

// Masking
$masker = new LogMasker();
$masker->mask(['password' => 'x', 'name' => 'ali']);
// ['password' => '***masked***', 'name' => 'ali']

// کد ارجاع
$ref = RefCodeGenerator::generate();  // 'AB12CD34'

// Renderers
use Tavaan\Engine\Error\Renderers\JsonRenderer;
use Tavaan\Engine\Error\Renderers\HtmlRenderer;
$json = (new JsonRenderer())->render(HttpException::notFound());
$html = (new HtmlRenderer())->render(HttpException::notFound());
```
